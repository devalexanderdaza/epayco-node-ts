### Test 2: README Method Usage Consistency Check

**Type:** Deterministic doc-surface consistency check

**File:** `tests/e1-s1-method-consistency.test.ts` (create new)

**What it validates:**

- Extracts all method references from README (e.g., `epayco.customers.create`)
- Compares against actual exported SDK surface
- Asserts all referenced methods exist in exports

**Expected behavior:**

```typescript
import { readFileSync } from "fs";
import { join } from "path";
import * as epayco from "../src/index";

describe("E1-S1: Method Surface Consistency", () => {
  it("should only document methods that exist in the SDK exports", () => {
    const readmeContent = readFileSync(
      join(process.cwd(), "README.md"),
      "utf-8",
    );

    // Extract documented method patterns: epayco.RESOURCE.METHOD
    const methodPattern = /epayco\.([\w]+)\.([\w]+)\(/g;
    const documentedMethods = new Set<string>();
    let match;
    while ((match = methodPattern.exec(readmeContent)) !== null) {
      documentedMethods.add(`${match[1]}.${match[2]}`);
    }

    // Get actual exported surface
    const actualSurface = new Set<string>();
    for (const [resourceName, resource] of Object.entries(epayco)) {
      if (typeof resource === "object" && resource !== null) {
        for (const methodName of Object.getOwnPropertyNames(resource)) {
          if (typeof (resource as any)[methodName] === "function") {
            actualSurface.add(`${resourceName}.${methodName}`);
          }
        }
      }
    }

    // Assert documented methods exist in surface
    for (const method of documentedMethods) {
      expect(actualSurface.has(method)).toBe(
        true,
        `README documents ${method} but it is not exported by the SDK`,
      );
    }

    // Assert no documented methods are missing
    expect(documentedMethods.size).toBeGreaterThan(0);
  });
});
```

**Pass criteria:**

- Test executes without error
- All documented methods pass existence check
- No false positives or false negatives

### Test 3: Export Snapshot Test (Regression Baseline)

**Type:** Deterministic snapshot test

**File:** `tests/e1-s1-export-snapshot.test.ts` (create new)

**What it validates:**

- Generates a deterministic representation of the SDK public surface
- Snapshots this representation for future regression detection
- Asserts the snapshot matches the baseline

**Expected behavior:**

```typescript
import { describe, it, expect } from "vitest";
import * as epayco from "../src/index";

describe("E1-S1: Export Surface Snapshot", () => {
  it("should have deterministic, documented public surface", () => {
    const surface: Record<string, string[]> = {};

    // Build surface map
    for (const [name, item] of Object.entries(epayco)) {
      if (typeof item === "object" && item !== null && !Array.isArray(item)) {
        surface[name] = Object.getOwnPropertyNames(item)
          .filter((m) => typeof (item as any)[m] === "function")
          .sort();
      }
    }

    // Each resource should have documented methods
    expect(surface).toMatchSnapshot();
  });
});
```

### Test 4: README Example Compilability (Spot Check)

**Type:** Integration validation test

**File:** `tests/e1-s1-readme-examples.test.ts` (create new)

**What it validates:**

- Extracts code block examples from README
- Validates syntactic structure (basic TypeScript/JS parse)
- Asserts no obvious structural issues

**Expected behavior:**

````typescript
import { readFileSync } from "fs";
import { join } from "path";

describe("E1-S1: README Examples Structure", () => {
  it("should have syntactically valid code examples in README", () => {
    const readmeContent = readFileSync(
      join(process.cwd(), "README.md"),
      "utf-8",
    );

    // Extract JavaScript/TypeScript code blocks
    const codeBlockPattern =
      /```(?:javascript|typescript|js|ts)?\n([\s\S]*?)\n```/g;
    let exampleCount = 0;
    let match;

    while ((match = codeBlockPattern.exec(readmeContent)) !== null) {
      const code = match[1];
      exampleCount++;

      // Basic parse check: should not have obviously unmatched braces
      const openBraces = (code.match(/{/g) || []).length;
      const closeBraces = (code.match(/}/g) || []).length;
      const openParens = (code.match(/\(/g) || []).length;
      const closeParens = (code.match(/\)/g) || []).length;

      expect(openBraces).toBe(
        closeBraces,
        `Example ${exampleCount}: unmatched braces`,
      );
      expect(openParens).toBe(
        closeParens,
        `Example ${exampleCount}: unmatched parentheses`,
      );
    }

    expect(exampleCount).toBeGreaterThan(0);
  });
});
````

**Pass criteria:**

- At least 5+ code examples found in README
- All examples pass structural validation
- No unmatched braces/parentheses
